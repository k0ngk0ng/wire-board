class Handler(BaseHTTPRequestHandler):
 def log_message(self,*args):pass
 def do_GET(self):
  route=urlsplit(self.path).path
  request={'path':route,'range':self.headers.get('Range'),'requested_at':time.monotonic()};requests.append(request)
  if route=='/':raw=b'<!doctype html><html lang="zh-CN"><meta charset="utf-8"><meta name="viewport" content="width=device-width,initial-scale=1"><title>Native audio API verification</title><style>body{font:16px system-ui;background:#f1f0e9;color:#263635;padding:20px;max-width:840px;margin:auto}button{font:inherit;background:#365e58;color:#fff;padding:12px;border:0;border-radius:6px}pre{white-space:pre-wrap}</style><h1>Native media verification</h1><button id="start">Run native media checks</button><pre>Ready</pre><script type="module" src="/harness.js"></script></html>';mime='text/html'
  elif route=='/blank':raw=b'<!doctype html><title>Native visibility check</title>Native visibility check';mime='text/html'
  elif route=='/harness.js':raw=(TMP/'harness.js').read_bytes();mime='text/javascript'
  elif route=='/slow/1f.wav':
   time.sleep(.35);raw=assets['/audio/1f.wav'].read_bytes();mime='audio/wav'
  elif route in assets:raw=assets[route].read_bytes();mime='audio/wav'
  else:self.send_error(404);return
  size=len(raw);first=0;last=size-1
  match=re.fullmatch(r'bytes=(\d+)-(\d*)',self.headers.get('Range',''))
  if match:first=int(match[1]);last=min(last,int(match[2])if match[2]else last)
  if first>last:self.send_error(416);return
  request['response_at']=time.monotonic()
  self.send_response(206 if match else 200);self.send_header('Content-Type',mime);self.send_header('Cache-Control','no-store');self.send_header('Content-Length',str(last-first+1));self.send_header('Accept-Ranges','bytes')
  if match:self.send_header('Content-Range',f'bytes {first}-{last}/{size}')
  self.end_headers()
  try:self.wfile.write(raw[first:last+1])
  except(BrokenPipeError,ConnectionResetError):pass
