// Failure cases are enumerated in web/FAILURES.md before implementation.
import test from 'node:test';
import assert from 'node:assert/strict';
import {InputTransport} from '../web/network-input.ts';
import {decodeDescriptor, resourceIDsForArena} from '../web/assets.ts';
const flush=async()=>{for(let i=0;i<12;i++)await Promise.resolve();};
function transport(){let now=0;const sent=[],pending=[];let resets=0;const input=new InputTransport({now:()=>now,inputWindow:()=>8,send:action=>{sent.push(action);return new Promise((resolve,reject)=>pending.push({resolve,reject}));},onReset:()=>resets++});const bind=(changes={})=>input.update({roomID:'r',userID:'u',seat:0,epoch:'e',serverAt:1000,receivedSequence:0,appliedSequence:0,canInput:true,...changes});bind();return{input,sent,pending,bind,setNow:value=>now=value,resets:()=>resets};}
test('MD twelve-bit short taps transmit in sequence without waiting for ACK',async()=>{const{input,sent,pending}=transport();const down=input.sample(0x740);const up=input.sample(0);assert.ok(down&&up&&down.sequence<up.sequence);assert.equal(sent.length,2);assert.deepEqual(sent.map(action=>action.buttons),[0x740,0]);pending[1].resolve();pending[0].resolve();await flush();assert.equal(input.pendingCount,0);input.dispose();});
test('cancellation has an independent lane and stale replies do not revive old input',async()=>{const{input,sent,pending,bind}=transport();input.sample(8);input.cancel();assert.equal(sent.length,2);assert.equal(sent[1].type,'release');assert.ok(sent[1].sequence>sent[0].sequence);bind({epoch:'next'});pending.forEach(value=>value.resolve());await flush();assert.equal(input.pendingCount,0);assert.equal(input.held,0);input.dispose();});
test('bounded input expires with release and never emits automation or seat claims',async()=>{const{input,sent,setNow}=transport();for(let i=0;i<64;i++)input.sample(i%2?0x800:0x400);assert.ok(input.pendingCount<=64);setNow(800);input.sample(8);assert.equal(input.held,0);assert.ok(sent.some(action=>action.type==='release'));for(const action of sent){assert.ok(!Object.hasOwn(action,'seat')&&!Object.hasOwn(action,'automated'));assert.equal(action.epoch,'e');}assert.throws(()=>input.sample(0x1000));input.dispose();});
test('received and applied ACKs are distinct and monotonic',()=>{const{input,bind}=transport();input.sample(0x100);input.sample(0);bind({receivedSequence:2,appliedSequence:0,serverAt:1100});assert.equal(input.unapplied.length,2);bind({receivedSequence:2,appliedSequence:1,serverAt:1200});assert.equal(input.unapplied.length,1);assert.throws(()=>bind({receivedSequence:1,appliedSequence:2,serverAt:1300}));input.dispose();});
test('only the selected arena and its required auxiliary asset are requested',()=>{assert.deepEqual(resourceIDsForArena(0),['rules','sprites','hud','render_meta','arena-default']);assert.deepEqual(resourceIDsForArena(10),['rules','sprites','hud','render_meta','arena-a','arena-a-aux']);assert.throws(()=>resourceIDsForArena(3));});
test('resource descriptor identity rejects malformed or foreign media mappings',()=>{assert.throws(()=>decodeDescriptor(new TextEncoder().encode('{"schema":1,"resources":[]}')));assert.throws(()=>decodeDescriptor(new TextEncoder().encode('null')));assert.throws(()=>decodeDescriptor(new TextEncoder().encode('{"schema":1,"schema":1,"resources":[]}')));});
