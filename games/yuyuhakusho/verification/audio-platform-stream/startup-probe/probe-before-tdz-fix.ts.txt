import { GameAudio } from '../../web/src/audio';
import type { GameAudioStreamHandle } from '../../web/src/game-api/audio-stream';
const h:any={ready:true,done:false,error:'',results:[],events:[],current:undefined,media:undefined,case:''};Object.assign(window,{streamStartupProbe:h});
const elements:HTMLAudioElement[]=[];const platform=new GameAudio(()=>{const m=new Audio();elements.push(m);const native=m.play.bind(m);m.play=()=>{h.events.push({kind:'play-call',case:h.case,at:performance.now(),time:m.currentTime,readyState:m.readyState});const p=native();p.then(()=>h.events.push({kind:'play-resolved',case:h.case,at:performance.now(),time:m.currentTime,readyState:m.readyState}),e=>h.events.push({kind:'play-rejected',case:h.case,at:performance.now(),error:String(e)}));return p;};for(const kind of ['playing','waiting','stalled','error','loadedmetadata','ended'])m.addEventListener(kind,()=>h.events.push({kind,case:h.case,at:performance.now(),time:m.currentTime,readyState:m.readyState,status:h.current?.status()}));return m;});platform.mount();const port=platform.channel('startup-probe') as any;
const sleep=(ms:number)=>new Promise(r=>setTimeout(r,ms));const snapshots=()=>{const m=h.media,s=h.current?.status();return{at:performance.now(),status:s,native:m?{readyState:m.readyState,networkState:m.networkState,paused:m.paused,ended:m.ended,time:m.currentTime,duration:Number.isFinite(m.duration)?m.duration:String(m.duration),preload:m.preload,buffered:Array.from({length:m.buffered.length},(_,i)=>[m.buffered.start(i),m.buffered.end(i)])}:null,playPending:(h.current as any)?.playPending,sourceDuration:String((h.current as any)?.source?.duration)};};
const send=async(row:any)=>{h.results.push(row);await fetch('/progress',{method:'POST',body:JSON.stringify({results:h.results,events:h.events,current:snapshots()})});};
async function fixture(name:string){const m=await fetch('/manifest/'+name).then(r=>r.json());const init=new Uint8Array(await fetch('/'+name+'/init').then(r=>r.arrayBuffer()));return{m,init,async piece(i:number){const f=m.fragments[i];return{sequence:f.sequence,startSample:f.start_sample,sampleFrames:f.sample_frames,data:new Uint8Array(await fetch('/'+name+'/fragment/'+i).then(r=>r.arrayBuffer()))};}};}
async function start(f:any,name:string,preload:string){h.case=name;h.current=port.openStream({playbackID:name,mimeType:f.m.mime,sampleRate:44100,initSegment:f.init,maxForwardSamples:441000,retainBackSamples:11025});if(!h.current)throw Error('open rejected');h.media=elements.at(-1);h.media.preload=preload;await h.current.ready;return h.current as GameAudioStreamHandle;}
document.querySelector<HTMLButtonElement>('#start')!.onclick=async()=>{try{await platform.unlock();for(const name of ['small-bulk','default-bulk'])for(const preload of ['none','auto']){const f=await fixture(name);const s=await start(f,name+'-'+preload,preload);let started=false;for(let i=0;i<Math.min(f.m.fragments.length,64);i++){const piece=await f.piece(i);if(!await s.append(piece))throw Error('valid diagnostic append rejected');await sleep(50);const row={case:h.case,append:i+1,frames:s.status().acceptedEndSample,...snapshots()};await send(row);if(h.media.currentTime>.01||h.events.some((e:any)=>e.case===h.case&&e.kind==='playing')){started=true;break;}}await send({case:h.case,caseComplete:true,started,final:snapshots()});s.stop();}
 const f=await fixture('small-bulk'),s=await start(f,'small-prefix-eos','none');for(let i=0;i<8;i++)await s.append(await f.piece(i));await sleep(300);await send({case:h.case,stage:'before-eos',...snapshots()});await s.end();await sleep(600);await send({case:h.case,stage:'after-eos',...snapshots()});s.stop();
 const steady=await start(f,'small-steady-low-ahead','auto');let cursor=0;while(h.media.currentTime<=.01&&cursor<64){await steady.append(await f.piece(cursor++));await sleep(15);}await send({case:h.case,stage:'startup-complete',cursor,...snapshots()});const start=performance.now();while(performance.now()-start<4000){const row=f.m.fragments[cursor];if(!row)break;if((row.start_sample+row.sample_frames)/44100-steady.status().currentTime<=.26){if(await steady.append(await f.piece(cursor)))cursor++;}await sleep(10);await send({case:h.case,stage:'steady',cursor,...snapshots()});}steady.stop();h.done=true;platform.dispose();await send({case:'complete'});}catch(e){h.error=String(e);platform.dispose();h.done=true;await send({case:'failure',error:h.error});}};
